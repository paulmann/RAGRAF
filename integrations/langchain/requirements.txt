# Зависимости интеграции RAGRAF ↔ LangChain (замерено 03.11.2026)
#
# Ставить в ОТДЕЛЬНОЕ окружение, а не в рабочее окружение RAGRAF: интеграция
# тянет langchain, а он — свой pydantic и свои протокольные пакеты.
#
#   python -m venv <окружение>
#   <окружение>\Scripts\python.exe -m pip install \
#       -r requirements.txt -r ../../requirements.txt
#
# Второй файл — рантайм самого RAGRAF: он нужен прямому вызову (фасад работает
# в том же процессе). MCP-пути он не нужен вовсе — там сервер живёт отдельно.

# --- LangChain и MCP ---
# Проверено на этих версиях: langchain 1.4.8, langchain-core 1.7.4.
langchain>=1.4,<1.5
langchain-core>=1.7,<1.8

# MCP-путь: пространство langchain.mcp появилось в langchain 1.4 и требует
# дополнение [mcp], которое тянет fastmcp (и вместе с ним mcp 2.x).
# Без дополнения импорт langchain.mcp падает на `No module named 'fastmcp'`.
langchain[mcp]>=1.4,<1.5

# ВНИМАНИЕ: старый пакет langchain-mcp-adapters 0.3.2 в этой среде НЕ
# импортируется — его код ждёт RequestContext из mcp.shared.context, а mcp 2.3.0
# его не отдаёт. Ставить его вместе с langchain[mcp] нельзя: получится пакет,
# который есть в окружении, но не работает. Нужен старый путь — фиксируйте
# mcp<2 и НЕ ставьте дополнение [mcp], это отдельная среда.

# --- тесты ---
pytest>=9.0,<10.0
